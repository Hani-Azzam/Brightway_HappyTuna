# Gateway dependencies. Self-contained: install into their own venv, separate
# from every other team's service.
#
#   python -m venv .venv && source .venv/bin/activate
#   pip install -r ceo_agent/requirements.txt
#
# All pins are exact so a rebuild cannot silently change MCP client behaviour.
#
# mcp is pinned to the last 1.x release. 2.0.0 shipped the same day and is now
# the default `pip install mcp`, but it removes streamablehttp_client, renames
# result fields to snake_case, and swaps httpx for httpx2 -- and its Client
# defaults to a discovery probe that the v1-era servers we consume (Node SDK
# 1.29.x, FastMCP 1.x) do not expect. We do not own those servers, so we stay on
# the protocol era they were built against. Migrating this package later is
# mechanical.
mcp==1.29.0
pydantic==2.13.4
PyYAML==6.0.3

# Only langchain-core, for StructuredTool in router.py -- the CEO agent will be
# LangGraph. Deliberately NOT langchain-mcp-adapters: it would open its own MCP
# sessions and hand the model tools directly, bypassing policy, identity
# injection, and the audit log, which is the entire reason this gateway exists.
#
# langchain, langgraph, and a model provider package are NOT pinned here. The
# gateway is model-agnostic and does not import them; whoever builds the agent
# picks them.
langchain-core==1.5.4

# --- tests ------------------------------------------------------------------
# pytest ceo_agent/tests/test_unit.py         needs nothing running
# pytest ceo_agent/tests/test_integration.py  needs the composed servers
pytest==9.1.1
pytest-asyncio==1.4.0

# httpx arrives as a transitive dependency of mcp 1.x and is imported directly by
# connection.py to classify connection failures. Left unpinned so mcp's own
# constraint governs the version.
# --- CEO agent runtime -------------------------------------------------------
# The LLM client (services/llm_client.py) speaks Gemini through LangChain.
# langchain-google-genai matches the 1.x langchain-core line pinned above; its
# ChatGoogleGenerativeAI(model=, api_key=, temperature=) surface -- the only
# part llm_client.py uses -- is stable across that range.
langchain-google-genai>=2.0.0
python-dotenv>=1.0.0,<2.0.0
